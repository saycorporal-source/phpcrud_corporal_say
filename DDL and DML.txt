CREATE TABLE milktea(
 id INT PRIMARY KEY,
 MilkteaID VARCHAR(50),
 MilkteaName VARCHAR(50)
);

CREATE TABLE customer(
 id INT PRIMARY KEY,
 CustomerID VARCHAR(50),
 CustomerName VARCHAR(50)
);

CREATE TABLE flavor(
 id INT PRIMARY KEY,
 FlavorID VARCHAR(50),
 FlavorName VARCHAR(50)
);


CREATE TABLE staff(
 id INT PRIMARY KEY,
 StaffID VARCHAR(50),
 StaffName VARCHAR(50)
);

CREATE TABLE orders (
    id INT AUTO_INCREMENT PRIMARY KEY,
    OrderID VARCHAR(50),
    CustomerID VARCHAR(50),
    MilkteaID VARCHAR(50),
    FlavorID VARCHAR(50),
    StaffID VARCHAR(50),
    Quantity INT
);
